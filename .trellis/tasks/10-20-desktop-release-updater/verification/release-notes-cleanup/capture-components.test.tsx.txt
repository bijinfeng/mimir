import { readFileSync, writeFileSync } from 'node:fs'
import { resolve } from 'node:path'
import { MimirProvider } from '@mimir/renderer/titlebar-kit'
import { act, createElement } from 'react'
import { createRoot } from 'react-dom/client'
import { expect, it } from 'vitest'
import { normalizeReleaseNotes } from '../../main/release-notes'
import { makeUpdateBridge } from './test-bridge'
import { UpdatePage } from './UpdatePage'

it('captures current real update component before and after note normalization', async () => {
  const dest = resolve(process.cwd(), '../../.trellis/tasks/10-20-desktop-release-updater/verification/release-notes-cleanup')
  const official = JSON.parse(readFileSync(resolve(dest, '../mac-official-manual/official-releases.json'), 'utf8')).newest
  expect(official.tag_name).toBe('v0.1.4')
  writeFileSync(resolve(dest, 'fixture.json'), JSON.stringify({ source: official.html_url, currentVersion: '0.1.3', version: '0.1.4', rawNotes: official.body, normalizedNotes: normalizeReleaseNotes(official.body), checkedAt: '2026-10-20T10:53:43.000Z', provenance: 'Official public release body; rendering is a component fixture, not a packaged release update.' }, null, 2))
  const { setThemeMode } = await import('../../../../../packages/renderer/src/theme/appearance')
  ;(globalThis as { IS_REACT_ACT_ENVIRONMENT?: boolean }).IS_REACT_ACT_ENVIRONMENT = true
  for (const theme of ['light', 'dark'] as const) {
    setThemeMode(theme)
    for (const width of [600, 360]) {
      for (const phase of ['before', 'after']) {
        const host = document.createElement('div')
        host.id = 'root'
        host.style.cssText = `width:${width}px;height:648px;display:flex;flex-direction:column;overflow:hidden`
        document.body.append(host)
        const root = createRoot(host)
        const { api } = makeUpdateBridge({ status: 'available', currentVersion: '0.1.3', version: '0.1.4', platform: 'darwin', arch: 'arm64', capability: 'manual', autoDownload: false, checkedAt: '2026-10-20T10:53:43.000Z', releaseNotes: phase === 'before' ? official.body : normalizeReleaseNotes(official.body) })
        await act(async () => root.render(createElement(MimirProvider, null, createElement(UpdatePage, { api }))))
        const text = host.querySelector('[aria-label="更新说明"]')?.textContent ?? ''
        expect(text.includes('mimir-desktop-build')).toBe(phase === 'before')
        expect(host.querySelector('[aria-label="更新说明"] a')).toBeNull()
        writeFileSync(resolve(dest, `${phase}-${theme}-${width}.html`), `<!doctype html><html lang="zh-CN" data-theme="${theme}" style="color-scheme:${theme}"><head><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1">${document.head.innerHTML}</head><body style="margin:0">${host.outerHTML}</body></html>`)
        await act(async () => root.unmount())
        host.remove()
      }
    }
  }
})
