  67. function ConciseNumberInput({
  68.   label,
  69.   value,
  70.   onChange,
  71.   min = 0,
  72.   max = 14400,
  73.   step = 1,
  74. }: NumberFieldProps) {
  75.   // Format only the view: unrelated edits must retain the exact source geometry.
  76.   const display = String(Number(value.toPrecision(6)))
  77.   const [draft, setDraft] = useState<string | null>(null)
  78.   const commit = () => {
  79.     setDraft(null)
  80.     if (draft === null || draft.trim() === '') return
  81.     const parsed = Number(draft)
  82.     if (!Number.isFinite(parsed) || parsed === Number(display)) return
  83.     const next = Math.min(max, Math.max(min, parsed))
  84.     if (next !== value) onChange(next)
  85.   }
  86.   return (
  87.     <InputNumber<string>
  88.       style={{ width: '100%' }}
  89.       aria-label={label}
  90.       title={label}
  91.       stringMode
  92.       value={draft ?? display}
  93.       min={String(min)}
  94.       max={String(max)}
  95.       step={String(step)}
  96.       controls={false}
  97.       // InputNumber otherwise expands tiny exponent values into long zero strings.
  98.       formatter={(_, info) => (info.userTyping ? info.input : (draft ?? display))}
  99.       onInput={setDraft}
  100.       onBlur={commit}
  101.       onKeyDown={(event) => {
  102.         if (event.key === 'Enter') commit()
  103.         if (event.key === 'Escape') setDraft(null)
  104.       }}
  105.     />
  106.   )

  750. function PropertyTools({ info, busy, act }: { info?: PdfLocalInfo; busy: boolean; act: Act }) {
  751.   const [metadata, setMetadata] = useState(
  752.     info?.metadata ?? { title: '', author: '', subject: '', keywords: '' },
  753.   )
  754.   const [outline, setOutline] = useState(
  755.     (info?.outline ?? []).map((e) => `${'  '.repeat(e.level)}${e.page} | ${e.title}`).join('\n'),
