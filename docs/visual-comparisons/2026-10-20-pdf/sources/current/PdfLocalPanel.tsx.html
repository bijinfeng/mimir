  68. function ConciseNumberInput({
  69.   label,
  70.   value,
  71.   onChange,
  72.   min = 0,
  73.   max = 14400,
  74.   step = 1,
  75.   disabled = false,
  76. }: NumberFieldProps) {
  77.   // Format only the view: unrelated edits must retain the exact source geometry.
  78.   const display = String(Number(value.toPrecision(6)))
  79.   const [draft, setDraft] = useState<string | null>(null)
  80.   const commit = () => {
  81.     if (disabled) return
  82.     setDraft(null)
  83.     if (draft === null || draft.trim() === '') return
  84.     const parsed = Number(draft)
  85.     if (!Number.isFinite(parsed) || parsed === Number(display)) return
  86.     const next = Math.min(max, Math.max(min, parsed))
  87.     if (next !== value) onChange(next)
  88.   }
  89.   return (
  90.     <InputNumber<string>
  91.       disabled={disabled}
  92.       style={{ width: '100%' }}
  93.       aria-label={label}
  94.       title={label}
  95.       stringMode
  96.       value={draft ?? display}
  97.       min={String(min)}
  98.       max={String(max)}
  99.       step={String(step)}
  100.       controls={false}
  101.       // InputNumber otherwise expands tiny exponent values into long zero strings.
  102.       formatter={(_, info) => (info.userTyping ? info.input : (draft ?? display))}
  103.       onInput={setDraft}
  104.       onBlur={commit}
  105.       onKeyDown={(event) => {
  106.         if (event.key === 'Enter') commit()
  107.         if (event.key === 'Escape') setDraft(null)
  108.       }}
  109.     />
  110.   )
