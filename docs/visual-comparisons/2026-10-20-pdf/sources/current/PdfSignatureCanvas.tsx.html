  288. function PdfSignatureCanvas({
  289.   refObject,
  290.   size,
  291.   draft,
  292.   basis,
  293.   busy,
  294.   onChange,
  295.   onCancel,
  296.   onApply,
  297. }: {
  298.   refObject: RefObject<SVGSVGElement | null>
  299.   size: { width: number; height: number }
  300.   draft: Draft
  301.   basis: SignatureRect
  302.   busy: boolean
  303.   onChange(rect: SignatureRect): void
  304.   onCancel(): void
  305.   onApply(): void
  306. }) {
  307.   const gesture = useRef<{
  308.       id: number
  309.       x: number
  310.       y: number
  311.       rect: SignatureRect
  312.       resize: boolean
  313.     } | null>(null),
  314.     callbacks = useRef({ onChange })
  315.   useEffect(() => {
  316.     callbacks.current = { onChange }
  317.   }, [onChange])
  318.   useEffect(() => {
  319.     const cancel = () => {
  320.       if (gesture.current) {
  321.         callbacks.current.onChange(gesture.current.rect)
  322.         gesture.current = null
  323.       }
  324.     }
  325.     window.addEventListener('blur', cancel)
  326.     window.addEventListener('pointercancel', cancel)
  327.     return () => {
  328.       window.removeEventListener('blur', cancel)
  329.       window.removeEventListener('pointercancel', cancel)
  330.     }
  331.   }, [])
  332.   const point = (e: PointerEvent): [number, number] => {
  333.     const r = refObject.current!.getBoundingClientRect()
  334.     return [
  335.       ((e.clientX - r.left) / r.width) * size.width,
  336.       ((e.clientY - r.top) / r.height) * size.height,
  337.     ]
  338.   }
  339.   const down = (e: PointerEvent, resize = false) => {
  340.     if (busy || gesture.current || e.button !== 0 || e.isPrimary === false) return
  341.     e.preventDefault()
  342.     e.stopPropagation()
  343.     refObject.current?.focus({ preventScroll: true })
  344.     const [x, y] = point(e)
  345.     if (!draft.rect) {
  346.       onChange(placeSignature(basis.width, basis.height, size, x, y))
  347.       return
  348.     }
  349.     refObject.current?.setPointerCapture?.(e.pointerId)
  350.     gesture.current = { id: e.pointerId, x, y, rect: draft.rect, resize }
  351.   }
  352.   const rect = draft.rect,
  353.     display = draft.display
  354.   return (
  355.     <svg
  356.       ref={refObject}
  357.       className="mimir-pdf-signature-canvas"
  358.       viewBox={`0 0 ${size.width} ${size.height}`}
  359.       preserveAspectRatio="none"
  360.       role="application"
  361.       aria-label="签名位置预览"
  362.       tabIndex={0}
  363.       onPointerDown={(e) => down(e)}
  364.       onPointerMove={(e) => {
  365.         const g = gesture.current
  366.         if (!g || g.id !== e.pointerId || busy) return
  367.         const [x, y] = point(e)
  368.         if (g.resize) {
  369.           const w = Math.max(8, g.rect.width + x - g.x)
  370.           const max = Math.min(
  371.             size.width - g.rect.x,
  372.             ((size.height - g.rect.y) * g.rect.width) / g.rect.height,
  373.           )
  374.           const width = Math.min(max, w)
  375.           onChange({ ...g.rect, width, height: (width * g.rect.height) / g.rect.width })
  376.         } else
  377.           onChange(
  378.             clampSignature({ ...g.rect, x: g.rect.x + x - g.x, y: g.rect.y + y - g.y }, size),
  379.           )
  380.       }}
  381.       onPointerUp={(e) => {
  382.         if (gesture.current?.id === e.pointerId) {
  383.           gesture.current = null
  384.           e.stopPropagation()
  385.           refObject.current?.releasePointerCapture?.(e.pointerId)
  386.         }
  387.       }}
  388.       onPointerCancel={() => {
  389.         if (gesture.current) onChange(gesture.current.rect)
  390.         gesture.current = null
  391.       }}
  392.       onKeyDown={(e) => {
  393.         if (e.key === 'Escape') {
  394.           e.preventDefault()
  395.           e.stopPropagation()
  396.           onCancel()
  397.         } else if (e.key === 'Enter' && rect) {
  398.           e.preventDefault()
  399.           e.stopPropagation()
  400.           onApply()
  401.         } else if (
  402.           rect &&
  403.           ['ArrowLeft', 'ArrowRight', 'ArrowUp', 'ArrowDown'].includes(e.key) &&
  404.           !busy
  405.         ) {
  406.           e.preventDefault()
  407.           const n = e.shiftKey ? 10 : 1
  408.           onChange(
  409.             clampSignature(
  410.               {
  411.                 ...rect,
  412.                 x: rect.x + (e.key === 'ArrowLeft' ? -n : e.key === 'ArrowRight' ? n : 0),
  413.                 y: rect.y + (e.key === 'ArrowUp' ? -n : e.key === 'ArrowDown' ? n : 0),
  414.               },
  415.               size,
  416.             ),
  417.           )
  418.         }
  419.       }}
  420.     >
  421.       {rect && (
  422.         <g>
  423.           <svg
  424.             x={rect.x}
  425.             y={rect.y}
  426.             width={rect.width}
  427.             height={rect.height}
  428.             viewBox={`0 0 ${basis.width} ${basis.height}`}
  429.             preserveAspectRatio="none"
  430.             opacity={0.85}
  431.           >
  432.             {display.preview ? (
  433.               <image
  434.                 href={`data:image/png;base64,${display.preview}`}
  435.                 width={basis.width}
  436.                 height={basis.height}
  437.                 preserveAspectRatio="none"
  438.               />
  439.             ) : (
  440.               display.strokes?.map((path, i) => (
  441.                 <path
  442.                   // Ordered immutable paths have no per-stroke IDs; identical strokes remain distinct.
  443.                   // eslint-disable-next-line react/no-array-index-key
  444.                   key={i}
  445.                   d={path.map(([x, y], j) => `${j ? 'L' : 'M'}${x},${y}`).join(' ')}
  446.                   fill="none"
  447.                   stroke={display.color}
  448.                   strokeWidth={display.thickness}
  449.                   strokeLinecap="round"
  450.                   strokeLinejoin="round"
  451.                 />
  452.               ))
  453.             )}
  454.           </svg>
  455.           <rect
  456.             x={rect.x}
  457.             y={rect.y}
  458.             width={rect.width}
  459.             height={rect.height}
  460.             className="mimir-pdf-signature-canvas__box"
  461.             vectorEffect="non-scaling-stroke"
  462.           />
  463.           <rect
  464.             x={rect.x + rect.width - 5}
  465.             y={rect.y + rect.height - 5}
  466.             width={10}
  467.             height={10}
  468.             className="mimir-pdf-signature-canvas__resize"
  469.             vectorEffect="non-scaling-stroke"
  470.             onPointerDown={(e) => down(e, true)}
  471.           />
  472.         </g>
  473.       )}
  474.     </svg>
  475.   )
  476. }
